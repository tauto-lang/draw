35fn logical(m: &WinitMessage): LogicalKey {
36  match msg_logical_kind(&m) {
37    1 => Character(msg_logical_text(&m))
38    2 => Named(msg_logical_text(&m))
39    3 => Dead(msg_logical_text(&m))
40    other => UnknownKey
41  }

58  pub fn scale(w: &mut WinitWindow): Result(u32, Text) { win_scale(&mut w) }
59  pub fn close(w: WinitWindow): Result((), Text) { win_close(w) }
60  pub fn pull(w: &mut WinitWindow, timeout_ms: u32): Result(Event, Text) {
61    match win_pull(&mut w, timeout_ms) {
62      Err(e) => Err(e)
63      Ok(m) => match msg_kind(&m) {
64        0 => Ok(Idle)
65        1 => Ok(CloseRequested)
66        2 => { let #(width, height) := msg_pair(&m); Ok(Resized(width, height)) }
67        3 => { Ok(ScaleChanged(msg_pair(&m).fst)) }
68        4 => Ok(RedrawRequested)
69        5 => Ok(FocusChanged(msg_flag(&m)))
70        6 => Ok(Key(msg_name(&m), logical(&m), msg_flag(&m), msg_repeat(&m)))
71        7 => { let #(x, y) := msg_position(&m); Ok(PointerMoved(x, y)) }
72        8 => Ok(PointerButton(msg_name(&m), msg_flag(&m)))
73        9 => Ok(PointerLeft)
74        10 => { let #(x, y) := msg_position(&m); Ok(WheelLines(x, y)) }
75        11 => { let #(x, y) := msg_position(&m); Ok(WheelPixels(x, y)) }
76        other => Err("unknown window event")
77      }
78    }
79  }
