54  pub fn create(d: &mut Device, bytes: u64, kind: BufferKind): Result(Buffer, Text) {
55    match make_buffer(&mut d, bytes, match kind { Storage => false, Uniform => true }) {

57      Err(e) => Err(e)
58    }
59  }

61    upload_buffer(&mut d, b.id, offset, &data)
62  }
63  pub fn close(d: &mut Device, b: Buffer): Result((), Text) { release(&mut d, b.id) }

123fn collect_bindings(xs: List(Binding), ids: &mut Vec(u64)): Unit {
124  match xs {
125    Nil => #()
126    Cons(x, tail) => {
127      Vec::push(&mut ids, match x { BindBuffer(b) => b.id, BindTexture(t) => t.id })
128      collect_bindings(tail, &mut ids)
129    }
130  }
131}

133pub module Pass {
134  pub fn submit_frame(d: &mut Device, p: Pass): Result(FrameStatus, Text) {
135    var ids: Vec(u64) := Vec::new()
136    collect_bindings(p.bindings, &mut ids)
137    match draw(&mut d, p.target.id, p.pipeline.id, &ids, p.clear_rgba, p.vertices, p.instances) {
138      Err(e) => Err(e)
139      Ok(0) => Ok(Submitted)
140      Ok(1) => Ok(Outdated)
141      Ok(2) => Ok(Lost)
142      Ok(3) => Ok(TimedOut)
143      Ok(_) => Err("unknown frame status")
144    }
145  }
146  pub fn submit(d: &mut Device, p: Pass): Result((), Text) {
147    match submit_frame(&mut d, p) {
148      Err(e) => Err(e)
149      Ok(Submitted) => Ok(#())
150      Ok(Outdated) => Err("surface needs reconfiguration")
151      Ok(Lost) => Err("surface lost")
152      Ok(TimedOut) => Err("surface acquisition timed out")
153    }
154  }
